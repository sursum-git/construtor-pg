Mapeamentos de importacao e exportacao

Contrato inicial da engine para fluxos entre entidades e arquivos. O TXT cobre leiaute posicional, leiaute por separador e estrutura hierarquica para registros pai, filhos e totalizadores. Agora tambem existe uma tela administrativa real com historico persistido, agendamentos e suporte a XML de exportacao e importacao declarativa.

Indice Construtor Demo visual Tela administrativa

Endpoints administrativos atuais:

GET  /api/admin/import-export-mappings
GET  /api/admin/import-export-mappings/{code}
GET  /api/admin/import-export-mappings/executions
GET  /api/admin/import-export-mappings/schedules
POST /api/admin/import-export-mappings
POST /api/admin/import-export-mappings/preview
POST /api/admin/import-export-mappings/execute
POST /api/admin/import-export-mappings/schedules
POST /api/admin/import-export-mappings/schedules/run-due

A tela administrativa real esta publicada em production/admin/import-export-mappings.html e no runtime por screenId=admin.integracoes. Agora existe tambem uma pagina local dedicada para smoke da UI completa em examples/pages/import-export-admin-demo.html, alem do comando php backend/bin/console app:import-export:run-schedules para execucao recorrente.

Exemplo 1: API -> tabela local

{
  "code": "api-produto-para-tabela",
  "name": "API produto para tabela",
  "direction": "import",
  "targetType": "entity",
  "targetCode": "produto_builder",
  "format": "entity_copy",
  "mapping": {
    "source": {
      "type": "entity",
      "entityCode": "api_crud_mock_produto",
      "alias": "api_produto",
      "mode": "list",
      "limit": 50
    },
    "destination": {
      "type": "entity",
      "entityCode": "produto_builder",
      "operation": "upsert",
      "matchBy": [
        { "targetField": "descricao", "sourcePath": "descricao" }
      ]
    },
    "fieldMappings": [
      { "sourcePath": "nome", "targetPath": "descricao" },
      { "sourcePath": "ativo", "targetPath": "ativo" },
      { "constant": "Importado por mapeamento", "targetPath": "observacao" }
    ]
  }
}

Exemplo 4: XML hierarquico com namespace, atributos e filhos repetitivos

{
  "format": "xml",
  "mapping": {
    "sources": [
      { "type": "entity", "entityCode": "cliente", "alias": "cliente", "mode": "list" },
      { "type": "entity", "entityCode": "cidade", "alias": "cidade", "mode": "list" }
    ],
    "destination": {
      "type": "file",
      "fileFormat": "xml",
      "fileNamePattern": "clientes_rico",
      "rootName": "doc:clientes",
      "prettyPrint": true,
      "namespaces": [
        { "prefix": "doc", "uri": "urn:demo:doc" }
      ],
      "rootAttributes": [
        { "name": "versao", "constant": "1.0" }
      ],
      "xmlLayouts": [
        {
          "name": "doc:cliente",
          "sourceAlias": "cliente",
          "attributes": [
            { "name": "id", "sourcePath": "id" },
            { "name": "status", "sourcePath": "status" }
          ],
          "fields": [
            { "name": "nome", "sourcePath": "nome" }
          ],
          "children": [
            {
              "name": "doc:cidade",
              "sourceAlias": "cidade",
              "linkBy": [
                { "parentPath": "id", "childField": "cliente_id" }
              ],
              "textSourcePath": "nome"
            }
          ]
        }
      ]
    }
  }
}

Exemplo 5: XML -> tabela local por XPath declarativo

{
  "format": "entity_copy",
  "mapping": {
    "source": {
      "type": "file",
      "fileFormat": "xml",
      "recordPath": "/doc:clientes/doc:cliente",
      "contentParameter": "xmlContent",
      "namespaces": [
        { "prefix": "doc", "uri": "urn:demo:doc" }
      ],
      "fields": [
        { "targetField": "codigo_externo", "xpath": "@id" },
        { "targetField": "status", "xpath": "@status" },
        { "targetField": "nome", "xpath": "doc:nome/text()" }
      ]
    },
    "destination": {
      "type": "entity",
      "entityCode": "cliente",
      "operation": "upsert",
      "matchBy": [
        { "targetField": "codigo_externo", "sourcePath": "codigo_externo" }
      ]
    },
    "fieldMappings": [
      { "sourcePath": "codigo_externo", "targetPath": "codigo_externo" },
      { "sourcePath": "nome", "targetPath": "nome" },
      { "sourcePath": "status", "targetPath": "status" }
    ]
  }
}

Exemplo 8: TXT hierarquico com TreeView, filho e totalizador

{
  "format": "txt_layout",
  "mapping": {
    "sources": [
      {
        "type": "entity",
        "entityCode": "nota_fiscal",
        "alias": "nf",
        "mode": "list"
      },
      {
        "type": "entity",
        "entityCode": "nota_fiscal_item",
        "alias": "item",
        "mode": "list"
      }
    ],
    "destination": {
      "type": "file",
      "fileFormat": "txt_layout",
      "layoutMode": "tree",
      "fileNamePattern": "sped_bloco_c",
      "recordLayouts": [
        {
          "nodeType": "record",
          "recordType": "C100",
          "label": "Nota fiscal",
          "sourceAlias": "nf",
          "lineMode": "delimited",
          "separator": "|",
          "fields": [
            { "constant": "C100" },
            { "sourcePath": "id" },
            { "sourcePath": "numero" },
            { "sourcePath": "cliente_nome" }
          ],
          "children": [
            {
              "nodeType": "record",
              "recordType": "C170",
              "label": "Itens da nota",
              "sourceAlias": "item",
              "lineMode": "delimited",
              "separator": "|",
              "linkBy": [
                { "parentPath": "id", "childField": "nota_id" }
              ],
              "fields": [
                { "constant": "C170" },
                { "sourcePath": "nota_id" },
                { "sourcePath": "produto_nome" },
                { "sourcePath": "quantidade" },
                { "sourcePath": "valor_total" }
              ]
            },
            {
              "nodeType": "totalizer",
              "recordType": "C190",
              "label": "Totalizador da nota",
              "sourceAlias": "item",
              "lineMode": "delimited",
              "separator": "|",
              "linkBy": [
                { "parentPath": "id", "childField": "nota_id" }
              ],
              "aggregates": [
                { "name": "quantidade_itens", "type": "count" },
                { "name": "valor_total", "type": "sum", "sourcePath": "valor_total" }
              ],
              "fields": [
                { "constant": "C190" },
                { "sourcePath": "_parent.id" },
                { "sourcePath": "_summary.quantidade_itens" },
                { "sourcePath": "_summary.valor_total", "transforms": [{ "type": "number_format", "decimals": 2, "decimalSeparator": ",", "thousandSeparator": \"\" }] }
              ]
            }
          ]
        }
      ]
    }
  }
}
